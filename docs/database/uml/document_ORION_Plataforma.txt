package "002 - Usuários e Perfis" {
    entity "perfil" as perfil {
        * id : uuid <<PK>>
        --
        * nome : varchar(100) <<UQ>>
        descricao : text
        * status : varchar(30) = 'ATIVO'
        * data_criacao : timestamptz
    }

    entity "usuario" as usuario {
        * id : uuid <<PK>>
        --
        * perfil_id : uuid <<FK>>
        * nome : varchar(150)
        * email : varchar(254) <<UQ>>
        telefone : varchar(30)
        * senha_hash : text
        documento : varchar(20) <<UQ>>
        tipo_documento : varchar(10)
        * status : varchar(30) = 'ATIVO'
        ultimo_acesso : timestamptz
        * data_criacao : timestamptz
        * data_atualizacao : timestamptz
    }
}
